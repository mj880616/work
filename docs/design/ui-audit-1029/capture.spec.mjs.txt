export async function metrics(page){
  return page.evaluate(()=>{
    const W=innerWidth,H=innerHeight;
    const shown=e=>{const r=e.getBoundingClientRect();if(r.width<1||r.height<1)return false;const cs=getComputedStyle(e);return cs.visibility!=='hidden'&&cs.display!=='none'&&cs.opacity!=='0'&&r.bottom>0&&r.top<H&&r.right>0&&r.left<W};
    const nm=e=>(e.getAttribute('aria-label')||e.textContent||e.id||'').replace(/\s+/g,' ').trim().slice(0,30);
    const sel=e=>e.tagName.toLowerCase()+(e.id?'#'+e.id:'')+(typeof e.className==='string'&&e.className.trim()?'.'+e.className.trim().split(/\s+/).slice(0,2).join('.'):'');
    const btns=[...document.querySelectorAll('button,a.button,[role=button],summary,select,input:not([type=hidden]):not([type=checkbox]):not([type=radio]),textarea')].filter(shown);
    const big=btns.filter(b=>{const r=b.getBoundingClientRect();return b.tagName!=='TEXTAREA'&&r.height>36.5&&!b.closest('nav,.mobile-tabs,.nav-sub')}).map(b=>{const r=b.getBoundingClientRect(),cs=getComputedStyle(b);return {n:nm(b),s:sel(b),h:Math.round(r.height),w:Math.round(r.width),fs:cs.fontSize,fw:cs.fontWeight}});
    const filled=btns.filter(b=>{const cs=getComputedStyle(b);const bg=cs.backgroundColor;const m=bg.match(/\d+/g);if(!m)return false;const [r,g,bb,a=1]=m.map(Number);if(+a===0)return false;const l=(r*299+g*587+bb*114)/1000;return l<140&&b.tagName==='BUTTON'&&!b.closest('nav,.mobile-tabs')}).map(b=>nm(b));
    const heads=[...document.querySelectorAll('h1,h2,h3,.screen-title,.view-title,.card-title,.section-title')].filter(shown).filter(e=>!e.classList.contains('a11y-only')).map(e=>({t:nm(e),s:sel(e),fs:getComputedStyle(e).fontSize,top:Math.round(e.getBoundingClientRect().top)}));
    const hints=[...document.querySelectorAll('p,.hint,.muted,small,.empty,.help,.notice,.desc,[class*=hint],[class*=empty],[class*=guide]')].filter(shown).filter(e=>!e.closest('button')&&(e.textContent||'').trim().length>=14&&e.children.length<3).map(e=>({t:nm(e).slice(0,40),s:sel(e),fs:getComputedStyle(e).fontSize})).slice(0,15);
    const cards=[...document.querySelectorAll('#appView *')].filter(shown).filter(e=>{const cs=getComputedStyle(e);const pad=parseFloat(cs.paddingTop)+parseFloat(cs.paddingLeft);const bordered=parseFloat(cs.borderTopWidth)>0||cs.boxShadow!=='none';const r=e.getBoundingClientRect();return bordered&&pad>=28&&r.width>200&&r.height>60}).map(e=>{const cs=getComputedStyle(e);return {s:sel(e),pad:cs.padding,rad:cs.borderRadius,shadow:cs.boxShadow!=='none',w:Math.round(e.getBoundingClientRect().width)}}).slice(0,10);
    const fsHist={};for(const e of document.querySelectorAll('#appView *')){if(!shown(e))continue;const t=[...e.childNodes].some(n=>n.nodeType===3&&n.textContent.trim());if(!t)continue;const f=getComputedStyle(e).fontSize;fsHist[f]=(fsHist[f]||0)+1}
    const view=document.querySelector('#appView .view-panel:not(.hidden)')||document.querySelector('#appView main')||document.body;
    const first=[...view.querySelectorAll('*')].filter(shown).filter(e=>[...e.childNodes].some(n=>n.nodeType===3&&n.textContent.trim()))[0];
    const fixedBottom=[...document.querySelectorAll('body *')].filter(e=>{const p=getComputedStyle(e).position;return p==='fixed'&&shown(e)&&e.getBoundingClientRect().bottom>=H-2&&e.getBoundingClientRect().height<120}).map(e=>({s:sel(e),top:Math.round(e.getBoundingClientRect().top)}));

    return {W,H,scrollH:document.documentElement.scrollHeight,big,filled,heads,hints,cards,fsHist,firstText:first?{t:nm(first),top:Math.round(first.getBoundingClientRect().top)}:null,fixedBottom,dialogs};
  });
}
