async function open(page,s){
  const go=v=>page.evaluate(async v=>{try{await window.KPTUViewLoader?.load?.(v)}catch{} window.KPTURouter?.go?.(v)},v);
  if(s==='note'){await go('basket');await page.waitForTimeout(900);await page.evaluate(()=>{const b=document.querySelector('[data-basket-kind-nav="note"]');b?.click();});}
  else if(s==='weekly'){await go('home');await page.waitForTimeout(1200);await page.locator('[data-journal-mode="week"]').first().click().catch(()=>{});}
  else if(s==='calmonth'){await go('calendar');}
  else await go(s);
  await page.waitForTimeout(1800);await page.evaluate(()=>window.scrollTo(0,0));await page.waitForTimeout(200);
}
